# app/hookTimeout

Pushed when an app hook exceeds its configured timeout.

<ResponseField name="sessionId" type="string (UUID)">
  Branded AppSessionId
</ResponseField>

<ResponseField name="appId" type="string">
  The appId field.
</ResponseField>

<ResponseField name="hookName" type="before_message_delivery | before_dispatch | on_join | on_session_active | on_close">
  The hookName field.
</ResponseField>

<ResponseField name="timeoutMs" type="integer">
  The timeoutMs field.
</ResponseField>
